// overlay placement — reserved top bar + offset
export function applyPlacement(
  position: "bottom" | "top",
  reservedTop: number,
  offsetPx: number,
): number {
  if (position !== "top") return 0;

  const reserved = Math.max(0, Math.trunc(reservedTop));
  const offset = Math.max(0, Math.trunc(offsetPx));

  // the card stage owns its own vertical rhythm;
  // the compact strip only needs the desktop bar.
  return reserved + 14 + offset;
}
